3// TimelineMarquee — widget-independent state machine for the timeline's
4// marquee (multi-select) drag (#847).
5//
6// Header-only and widget-free by design, like TimelineInteractionPolicy.h:
7// the drag contract is reachable headless (AESTRA_CI=ON disables linking
8// AestraUI) so the two failure modes that presented as an app hang are pinned
9// by tests:
10// 1. An active marquee OWNS every mouse event until the button that began
11// it is released — the pointer can never escape a drag, and a foreign
12// button can never finalize someone else's gesture or move its band.
13// 2. Per-move work is pure bookkeeping (endpoint update only). This type
14// carries no invalidation, cache, or render surface at all, so a move
15// cannot trigger a per-move invalidation storm.
16//
17// The widget maps NUIMouseButton onto MarqueeButton tokens, clamps cursor
18// positions into the grid area, and routes an active marquee BEFORE sibling
19// controls (the minimap consumes in-bounds releases even when idle, which
20// would strand the gesture) — clamping and routing policy stay with the
21// widget, sequencing rules live here.

31using MarqueeButton = int;
32
36enum class MarqueeEventKind {
37 Move, // pointer move/drag: endpoint follows the position
38 Press, // button down: consumed, no state change
39 Release, // button up: endpoint follows, finalizes when initiating button
40 Other, // scroll, enter/leave, synthetic: consumed, no state change
41};
42
43struct TimelineMarqueeDrag {
46 bool begin(bool pressed, MarqueeButton btn, float x, float y, bool inTrackArea) {
47 if (!pressed || !inTrackArea || m_active || btn == 0) {
48 return false;
49 }
50 m_active = true;
51 m_button = btn;
52 m_startX = x;
53 m_startY = y;
54 m_endX = x;
55 m_endY = y;
56 return true;
57 }
58
60 bool ownsEvent() const { return m_active; }
61
66 bool onEvent(MarqueeEventKind kind, MarqueeButton btn, float x, float y) {
67 if (!m_active) {
68 return false;
69 }
70 if (kind == MarqueeEventKind::Release) {
71 if (btn != m_button) {
72 return false; // foreign release: consumed, endpoint untouched
73 }
74 m_endX = x;
75 m_endY = y;
76 return true;
77 }
78 if (kind == MarqueeEventKind::Move) {
79 m_endX = x;
80 m_endY = y;
81 }
82 return false; // Press and Other: consumed, no state change
83 }
84
86 void finalize() {
87 m_active = false;
88 m_button = 0;
89 }
90
91 // ── Read-only queries (consumers + tests) ──
92 bool active() const { return m_active; }
93 MarqueeButton button() const { return m_button; }
94 float startX() const { return m_startX; }
95 float startY() const { return m_startY; }
96 float endX() const { return m_endX; }
97 float endY() const { return m_endY; }
98
101 float rectMinX() const { return std::min(m_startX, m_endX); }
102 float rectMinY() const { return std::min(m_startY, m_endY); }
103 float rectMaxX() const { return std::max(m_startX, m_endX); }
104 float rectMaxY() const { return std::max(m_startY, m_endY); }
105 float rectWidth() const { return rectMaxX() - rectMinX(); }
106 float rectHeight() const { return rectMaxY() - rectMinY(); }
107
108private:
109 bool m_active = false;
110 MarqueeButton m_button = 0; // the button that began the drag
111 float m_startX = 0.0f;
112 float m_startY = 0.0f;
113 float m_endX = 0.0f;
114 float m_endY = 0.0f;
115};
