101 void attachBuffer(ClipSource* source, std::shared_ptr<AudioBufferData> buffer) {
102 if (!source) {
103 return;
104 }
105 // Reject invalid payloads before they can clobber ready content: a
106 // failed attach must never discard valid audio and its cache.
107 if (!buffer || !buffer->isValid()) {
108 return;
109 }
110 // First manager to attach owns the source; a later attach through a
111 // different manager still proceeds (single-manager process today),
112 // and this manager's revision is what the sweep watches.
113 source->setOwner(this);
114 const bool wasReady = source->isValid();
115 source->setBuffer(std::move(buffer)); // bumps itself on a readiness flip
116 if (wasReady && source->isValid()) {
117 ++m_revision; // ready -> ready replacement still discards the cache
118 }
119 }

179 bool relinkSource(ClipSourceID id, const std::string& newFilePath) {
180 if (newFilePath.empty()) {
181 return false;
182 }
183 auto it = m_sources.find(id.value);
184 if (it == m_sources.end()) {
185 return false;
186 }
187 auto taken = m_pathToId.find(newFilePath);
188 if (taken != m_pathToId.end() && taken->second != id) {
189 return false;
190 }
191 const std::string oldPath = it->second->getFilePath();
192 it->second->setFilePath(newFilePath);
193 if (oldPath != newFilePath) {
194 m_pathToId.erase(oldPath);
195 m_pathToId[newFilePath] = id;
196 ++m_revision;
197 }
198 return true;
199 }
