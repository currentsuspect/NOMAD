14void ClipSource::setBuffer(std::shared_ptr<AudioBufferData> buffer) {
15 const bool wasReady = isValid();
16 m_buffer = std::move(buffer);
17 m_waveformCache.reset();
18 ++m_contentRevision;
19 clearFilteredVariant(); // new content invalidates any anti-aliased copy
20
21 // INVARIANT: every not-ready -> ready transition must bump the owning
22 // SourceManager's revision. TrackManagerUI change-gates its "install
23 // missing waveform caches" sweep on that revision, so a source whose buffer
24 // arrives through a path that does not go through SourceManager (e.g. an
25 // async decode completing onto an already-deduped source) would otherwise
26 // render as a bare centre line until some unrelated source event happened
27 // to run the sweep. A null owner means standalone source: nothing to wake.
28 if (!wasReady && isValid() && m_owner) {
29 m_owner->bumpRevision();
30 }
31}
