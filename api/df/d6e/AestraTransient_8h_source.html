1// © 2026 Aestra Studios — All Rights Reserved. Licensed for personal & educational use only.
2// AestraTransient V1 — feed-forward transient shaper (envelope-follower attack/decay
3// gain scaling). Zero latency: output[i] depends only on inputs up to i, no
4// lookahead, no reported latency, no PDC.
5//
6// Signal path (mono-linked detector, per-channel gain application):
7// in -> |x| -> fast envelope (peaks) + slow envelope (body)
8// -> normalized difference -> tanh-bounded gain computer
9// -> smoothed gain -> output trim -> mix with dry
10//
11// The fast envelope tracks onsets within ~1 ms while the slow envelope holds the
12// steady body, so their normalized difference is positive on an attack portion
13// and negative on a decay portion. The Attack amount scales the gain on the
14// attack portion, the Sustain amount scales it on the decay portion; both are
15// bipolar around neutral.
16

49 // Detector time constants (seconds). Fixed by design: the workhorse brief
50 // exposes only the two amounts, not the detector tuning.
51 static constexpr float kFastAttackSec = 0.001f;
52 static constexpr float kFastReleaseSec = 0.008f;
53 static constexpr float kSlowAttackSec = 0.025f;
54 static constexpr float kSlowReleaseSec = 0.120f;
55 static constexpr float kMaxSwingDb = 12.0f;
56 static constexpr float kGainSmoothSec = 0.002f;
57 static constexpr float kEnvFloor = 1.0e-6f;

89 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
90 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) override {
91 (void)midiInput;
92 (void)midiOutput;
93
94 // Zero reported latency means internal bypass is a plain copy.
95 if (!m_active.load(std::memory_order_relaxed) || m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
96 copyOrClear(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
97 m_wasBypassed = true;
98 return;
99 }
100 if (m_wasBypassed) {
101 // Flush stale envelope/gain state from before the bypass so
102 // reactivation does not emit it.
103 resetRuntimeState();
104 snapSmoothedParams();
105 m_wasBypassed = false;
106 }
107
108 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
109 const bool stereo = channels >= 2;
110
111 float blockInputPeak = 0.0f;
112 float blockOutputPeak = 0.0f;
113
114 float fast = m_fast;
115 float slow = m_slow;
116 float gainSm = m_gainSmoothed;
117
118 for (uint32_t blockStart = 0; blockStart < numFrames; blockStart += kCtrlBlock) {
119 const uint32_t blockEnd = std::min(blockStart + kCtrlBlock, numFrames);
120
121 // Smoothing coefficient derives from the actual segment length so
122 // the 2 ms time constant holds for partial segments and short
123 // host callbacks alike (block-size independent).
124 const float segSmoothCoeff = 1.0f - std::exp(-static_cast<float>(blockEnd - blockStart) /
125 (0.002f * static_cast<float>(m_sampleRate)));
126 m_attackSmoothed += (getParameter(kAttack) - m_attackSmoothed) * segSmoothCoeff;
127 m_sustainSmoothed += (getParameter(kSustain) - m_sustainSmoothed) * segSmoothCoeff;
128 m_outputSmoothed += (getParameter(kOutput) - m_outputSmoothed) * segSmoothCoeff;
129 m_mixSmoothed += (getParameter(kMix) - m_mixSmoothed) * segSmoothCoeff;
130
131 const float amtA = bipolarFromNorm(m_attackSmoothed);
132 const float amtS = bipolarFromNorm(m_sustainSmoothed);
133 const float outGain = dbToLinear(outputDbFromNorm(m_outputSmoothed));
134 const float wet = std::clamp(m_mixSmoothed, 0.0f, 1.0f);
135 const float dry = 1.0f - wet;
136
137 for (uint32_t i = blockStart; i < blockEnd; ++i) {
138 const float inL = sanitizeSample(readInput(inputs, numInputChannels, 0, i));
139 const float inR = stereo ? sanitizeSample(readInput(inputs, numInputChannels, 1, i)) : inL;
140 blockInputPeak = std::max(blockInputPeak, std::max(std::abs(inL), std::abs(inR)));
141
142 const float abs = std::max(std::abs(inL), std::abs(inR));
143 fast += (abs - fast) * (abs > fast ? m_fastAttackCoeff : m_fastReleaseCoeff);
144 slow += (abs - slow) * (abs > slow ? m_slowAttackCoeff : m_slowReleaseCoeff);
145
146 // Normalized envelope difference: positive on an attack
147 // portion, negative on a decay portion, bounded via tanh.
148 const float diff = (fast - slow) / std::max(slow, kEnvFloor);
149 const float shaped = std::tanh(diff);
150 const float gainDb = (diff >= 0.0f ? amtA * shaped : amtS * (-shaped)) * kMaxSwingDb;
151 const float gain = dbToLinear(gainDb);
152
153 // One-pole smoothing of the applied gain prevents zipper noise
154 // when the gain computer jumps between attack and decay.
155 gainSm += (gain - gainSm) * m_gainSmoothCoeff;
156 if (!std::isfinite(gainSm) || gainSm < 1.0e-12f)
157 gainSm = 1.0f; // NaN guard + denormal flush; unity is the safe state
158
159 const float outL = flushDenormal((inL * gainSm * outGain) * wet + inL * dry);
160 const float outR = flushDenormal((inR * gainSm * outGain) * wet + inR * dry);
161 blockOutputPeak = std::max(blockOutputPeak, std::max(std::abs(outL), std::abs(outR)));
162
163 if (numOutputChannels > 0 && outputs[0])
164 outputs[0][i] = outL;
165 if (numOutputChannels > 1 && outputs[1])
166 outputs[1][i] = stereo ? outR : outL;
167 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
168 if (outputs[ch])
169 outputs[ch][i] = sanitizeSample(readInput(inputs, numInputChannels, ch, i));
170 }
171 }
172 }
173
174 m_fast = flushEnvelope(fast);
175 m_slow = flushEnvelope(slow);
176 m_gainSmoothed = gainSm;
177 m_inputLevel.store(blockInputPeak, std::memory_order_relaxed);
178 m_outputLevel.store(blockOutputPeak, std::memory_order_relaxed);
179 }

207 std::string getParameterDisplay(uint32_t id) const override {
208 if (id >= kParamCount)
209 return "";
210 const float v = getParameter(id);
211 char buf[32];
212 switch (id) {
213 case kAttack: {
214 const float amt = bipolarFromNorm(v);
215 std::snprintf(buf, sizeof(buf), "%+.0f%%", amt * 100.0f);
216 return buf;
217 }
218 case kSustain: {
219 const float amt = bipolarFromNorm(v);
220 std::snprintf(buf, sizeof(buf), "%+.0f%%", amt * 100.0f);
221 return buf;
222 }
223 case kOutput:
224 std::snprintf(buf, sizeof(buf), "%+.1f dB", outputDbFromNorm(v));
225 return buf;
226 case kMix:
227 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
228 case kBypass:
229 return v > 0.5f ? "ON" : "OFF";
230 default:
231 return "";
232 }

235 std::vector<uint8_t> saveState() const override {
236 struct Blob {
237 uint32_t magic = kStateMagic;
238 uint32_t version = 1;
239 float params[kParamCount] = {};
240 } blob;
241 for (uint32_t i = 0; i < kParamCount; ++i) {
242 blob.params[i] = getParameter(i);
243 }
244 const auto* data = reinterpret_cast<const uint8_t*>(&blob);
245 return {data, data + sizeof(blob)};
246 }
247
248 bool loadState(const std::vector<uint8_t>& state) override {
249 if (state.size() < sizeof(uint32_t) * 2)
250 return false;
251 uint32_t magic = 0;
252 std::memcpy(&magic, state.data(), sizeof(magic));
253 if (magic != kStateMagic)
254 return false;
255 struct Blob {
256 uint32_t magic;
257 uint32_t version;
258 float params[kParamCount];
259 };
260 if (state.size() < sizeof(Blob))
261 return false;
262 Blob blob{};
263 std::memcpy(&blob, state.data(), sizeof(blob));
264 if (blob.version < 1 || blob.version > 1)
265 return false;
266 // Reject the whole blob before touching any parameter: a corrupt
267 // value must not leave the instance half-updated (AGENTS.md §12).
268 for (uint32_t i = 0; i < kParamCount; ++i) {
269 if (!std::isfinite(blob.params[i]))
270 return false;
271 }
272 for (uint32_t i = 0; i < kParamCount; ++i) {
273 setParameter(i, blob.params[i]);
274 }
275 return true;
276 }

299 // ── Parameter mapping (public: shared with editor/tests) ──
300 static float bipolarFromNorm(float value) { return std::clamp(value, 0.0f, 1.0f) * 2.0f - 1.0f; }
301
302 static float outputDbFromNorm(float value) { return -12.0f + std::clamp(value, 0.0f, 1.0f) * 24.0f; }

305 static constexpr uint32_t kCtrlBlock = 16; // control-rate interval for parameter smoothing

307 void resetRuntimeState() {
308 m_fast = 0.0f;
309 m_slow = 0.0f;
310 m_gainSmoothed = 1.0f;
311 m_wasBypassed = false;
312 const float sr = static_cast<float>(m_sampleRate);
313 // Per-sample one-pole coefficients from the fixed detector constants.
314 m_fastAttackCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kFastAttackSec));
315 m_fastReleaseCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kFastReleaseSec));
316 m_slowAttackCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kSlowAttackSec));
317 m_slowReleaseCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kSlowReleaseSec));
318 m_gainSmoothCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kGainSmoothSec));
319 m_inputLevel.store(0.0f, std::memory_order_relaxed);
320 m_outputLevel.store(0.0f, std::memory_order_relaxed);
321 }

330 static float flushEnvelope(float value) { return (!std::isfinite(value) || value < 1.0e-12f) ? 0.0f : value; }

332 // ── Utility ──
333 static void copyOrClear(const float* const* inputs, float** outputs, uint32_t numInputChannels,
334 uint32_t numOutputChannels, uint32_t numFrames) {
335 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
336 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
337 if (inputs[ch] != outputs[ch]) {
338 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
339 }
340 } else if (outputs[ch]) {
341 std::memset(outputs[ch], 0, numFrames * sizeof(float));
342 }
343 }
344 }

346 static float readInput(const float* const* inputs, uint32_t channels, uint32_t channel, uint32_t frame) {
347 if (channel >= channels || !inputs[channel]) {
348 return (channel == 1 && channels > 0 && inputs[0]) ? inputs[0][frame] : 0.0f;
349 }
350 return inputs[channel][frame];
351 }

353 static float sanitizeSample(float sample) {
354 if (!std::isfinite(sample))
355 return 0.0f;
356 return std::clamp(sample, -16.0f, 16.0f);
357 }

359 static float dbToLinear(float db) { return std::exp(db * 0.11512925464970229f); }
360
361 static float flushDenormal(float value) { return std::abs(value) < 1.0e-20f ? 0.0f : value; }
