68inline double timelineGridOffsetToBeat(float offsetFromGridStart, float scrollOffset, float pixelsPerBeat) {
69 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
70 return 0.0;
71 }
72 return (static_cast<double>(offsetFromGridStart) + static_cast<double>(scrollOffset)) /
73 static_cast<double>(pixelsPerBeat);
74}

77inline float timelineBeatToGridOffset(double beat, float scrollOffset, float pixelsPerBeat) {
78 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
79 return 0.0f;
80 }
81 return static_cast<float>(beat * static_cast<double>(pixelsPerBeat)) - scrollOffset;
82}

115inline float safeClampFloat(float value, float a, float b) {
116 // Bounds first — "in range" is meaningless without a range.
117 if (!std::isfinite(a) && !std::isfinite(b))
118 return 0.0f;
119 if (!std::isfinite(a))
120 a = b;
121 if (!std::isfinite(b))
122 b = a;
123
124 const float lo = std::min(a, b);
125 const float hi = std::max(a, b);
126
127 // NaN compares false against everything, so it would otherwise fall through
128 // both bounds checks and be returned unchanged.
129 if (std::isnan(value))
130 return lo;
131
132 // These two also absorb -inf and +inf, which order correctly against finite
133 // bounds and so need no special case of their own.
134 if (value <= lo)
135 return lo;
136 if (value >= hi)
137 return hi;
138 return value;
139}
